# Allowlist for tools/decode_stream.py check, v11 (Mali-G615), vs runs/20261115-cmf-libkbase/panvk-trace/draw-flat.jsonl (PanVK flat triangle).
# Format:  KEY [ours=0xV] [golden=0xV]   # justification.  KEY is a glob over "<run>.<desc>.<field>" (run = idvs|frag). Values pin the exact known difference,
# so a DIFFERENT wrong value still fails. VAs/pointers, fence/queue/CQS addresses are already ignored by the tool (any qword holding a GPU VA -> 'VA').
# Every line here is a difference that exists today AND where our triangle renders correctly on the CMF (t_triangle PASS, README.md); not proven equivalent in general.

# --- shader-binary dependent: ours are the compiler-spike/own shaders, PanVK's are its own compile of tests/t.vert,t.frag ---
idvs.FAUv.cnt                 ours=6 golden=4    # vertex FAU word count follows our shader's constant layout (compiler-spike pos.vert, 6 words with 0.5 and 0x10)
idvs.FAUv.has_3f000000        ours=1             # constant 0.5f used by our vertex shader (PanVK shader has none)
idvs.FAUv.has_00000010        ours=1             # constant 0x10 used by our vertex shader (varying stride/offset), PanVK shader has none
idvs.FAUv.has_3f800000        golden=1           # constant 1.0f present in PanVK vertex FAU only; our shader derives it differently
idvs.FAUf.cnt                 ours=6 golden=5    # fragment FAU word count differs with the shader
idvs.FAUf.has_00000019        golden=1           # RT conversion words (0x19, 0xed000) are in PanVK's FAU; our FS reads them from the Blend descriptor (w2,w3 = same values)
idvs.FAUf.has_000ed000        golden=1           # see above; if the Blend descriptor ever loses 0x000ed000 the pixfmt688 self-test catches it via Blend w3
idvs.SPD_pos.w2               ours=0x20000000 golden=0x1000   # shader program descriptor preload/flags: property of the binary
idvs.SPD_vary.w2              ours=0x20000100 golden=0x1100   # same
idvs.SPD_frag.w2              ours=0x20000200 golden=0x1180   # same
idvs.SPD_*.w3                 ours=0x0 golden=0x8000          # golden-only SPD word 3 (0x8000); unidentified; our shaders run correctly without it on the CMF

# --- known open since r17, triangle renders fine anyway (PANVK-SEQUENCE.md) ---
idvs.r48                      ours=0x20 golden=0x10           # IDVS r48 (varying size/stride): PANVK-SEQUENCE.md lists it as a known difference; not root-caused

# --- framebuffer: PanVK renders to an optimal-layout image with ZS/CRC extension; we render to a linear dma-buf ---
frag.FBD.tag                  ours=1 golden=3                 # FBD ptr tag bit 1 = ZS/CRC ext block; we have none (TRIANGLE-NEXT.md:50)
frag.FBD.w12                  ours=0x20010000 golden=0xf0210000  # bit 21 'has ZS CRC ext' (see tag) + bits 31,30,28 which PanVK sets and we do not; unidentified
frag.RT.w1                    ours=0x86880299 golden=0x86888d99  # RT block format bits 11:8: ours 2 = linear (dma-buf target), PanVK 0xd = its image layout
frag.RT.w2                    ours=0x0 golden=0x2801             # PanVK image-layout/AFBC-related word, n/a for linear
frag.RT.w10                   ours=0x400 golden=0x800            # row stride: ours 256 px * 4 B linear; PanVK tiled layout
frag.RT.w11                   ours=0x40000 golden=0x1000         # surface stride, same reason
